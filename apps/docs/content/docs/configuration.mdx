All options live in `lintmax.config.ts`.

## Top-level options

### ignores

```ts
ignores: ['packages/ui/**', '.intlayer/cache/**']
```

### compact

Controls consecutive blank line removal in source files (default `true`).

```ts
compact: false
```

### comments

Controls comment deletion (default `true`).

```ts
comments: false
```

### tailwind

```ts
tailwind: false
```

```ts
tailwind: 'src/styles/globals.css'
```

When omitted or `true`, lintmax auto-detects common entries. In monorepos, it prefers `ui/src/styles/globals.css` when multiple candidates exist.

### tsconfigRootDir

```ts
tsconfigRootDir: '/absolute/project/root'
```

## eslint

Use top-level `tailwind` and `tsconfigRootDir` instead of nesting them.

### eslint.off

```ts
eslint: {
  off: ['@typescript-eslint/no-magic-numbers']
}
```

### eslint.ignores

```ts
eslint: {
  ignores: ['vendor/**']
}
```

### eslint.append

For JSON-serializable inline config:

```ts
eslint: {
  append: [{ files: ['tests/**'], rules: { 'no-magic-numbers': 'off' } }]
}
```

For preset modules with runtime plugins, use `eslintImport(...)`:

```ts
import { defineConfig, eslintImport } from 'lintmax'

export default defineConfig({
  eslint: {
    append: [
      eslintImport({
        files: ['backend/convex/**/*.ts'],
        from: '@noboil/convex/eslint',
        name: 'recommended'
      })
    ]
  }
})
```

## biome

### biome.off

```ts
biome: {
  off: ['noConsole']
}
```

### biome.ignores

```ts
biome: {
  ignores: ['_generated/**']
}
```

### biome.overrides

```ts
biome: {
  overrides: [{ includes: ['packages/ui/**'], off: ['noConsole'] }]
}
```

## oxlint

### oxlint.off

```ts
oxlint: {
  off: ['no-console']
}
```

### oxlint.ignores

```ts
oxlint: {
  ignores: ['_generated/**']
}
```

### oxlint.overrides

```ts
oxlint: {
  overrides: [{ files: ['**/*.test.ts'], off: ['no-console'] }]
}
```

## Shared file-pattern overrides

```ts
overrides: {
  '**/*.test.ts': {
    eslint: ['no-console'],
    oxlint: ['no-console'],
    biome: ['noConsole']
  }
}
```

Each entry must define at least one linter action: `biome`, `eslint`, or `oxlint`.

## TypeScript preset

```json
{ "extends": "lintmax/tsconfig" }
```
